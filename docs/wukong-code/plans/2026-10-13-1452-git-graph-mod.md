# cc-git-graph Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use wukong-code:subagent-driven-development (recommended) or wukong-code:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 Claude Code 桌面应用里交付一个只读的 Git 图谱看板 mod:输入 `/git-graph` 打开面板,显示泳道图、提交列表、分支筛选,点击提交展开详情。

**Architecture:** 三层:`git` 层(只读命令封装与输出解析)、`layout` 层(泳道布局与单行 SVG,纯函数)、`ui/state` 层(`atom` 状态、控制器、`ui.render` 渲染)。每行由 `Svg`(图谱列)加 `Button`(描述与徽章,点击热区)加三个 `Text` 列组成。

**Tech Stack:** Claude Code mod(`claude-code` API 2.1.295,JSX 编译到全局 `h`)、TypeScript strict、`claude plugin test`(`claude-code/testing`)、`claude plugin validate`、`tsc`。

**Requirements:** [PRD.md](../../../PRD.md) v1.1。**Design:** [技术设计 spec](../specs/2026-10-13-1447-git-graph-mod-technical-design.md)。

## Global Constraints

- 目标 surface 只有 `desktop`;其他 surface 只显示一句提示。
- 所有 git 命令只读,子命令必须在白名单 `log`、`show`、`for-each-ref`、`rev-parse` 内;只通过 `hooks/git/run-git.ts` 调用 `$.process.run`。
- 每页 **50** 条提交(`PAGE_SIZE = 50`),多取 1 条判断 `hasMore`。
- 分支下拉单选,`'*'` 表示 Show All;不含搜索、自动刷新、多仓库、树形文件视图。
- 提交文本一律按纯文本渲染;SVG 字符串里只拼数字和常量颜色。
- 状态更新一律返回新对象,不原地修改。
- 单文件目标 200–400 行,上限 800 行;函数小于 50 行。
- 模块只用静态 `import`,不用 `require` 和 `import()`。
- 仓库根目录就是 mod:`.claude-plugin/plugin.json`、`.claude-plugin/marketplace.json`、`hooks/`、`types/`。
- 提交信息格式 `<type>: <description>`,并以 `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>` 结尾。

- [ ] **Step 3: 写探针 `hooks/register.tsx`**

探针只画 300 行,每行放 4 个不同高度(18/20/22/24 像素)的竖线 SVG 和一个 `Button`,并带两个滚动按钮:

```tsx
import type { ElementTable, Register } from 'claude-code'

const PANE = 'git-graph'
const HEIGHTS = [18, 20, 22, 24] as const
const ROWS = 300

const lineSvg = (height: number): string =>
  `<svg xmlns="http://www.w3.org/2000/svg" width="14" height="${height}" viewBox="0 0 14 ${height}">` +
  `<line x1="7" y1="0" x2="7" y2="${height}" stroke="#4f8ef7" stroke-width="2"/>` +
  `<circle cx="7" cy="${height / 2}" r="4" fill="#4f8ef7"/></svg>`

export const register: Register = on => {
  on('session.start', async ($, e, next) => {
    await $.command.register({ name: 'git-graph', description: 'Probe: Git graph board' })

    return next(e)
  })

  on('command.run', { command: 'git-graph' }, async $ => {
    await $.ui.open({ id: PANE, title: 'Git Graph probe' })

    return { text: 'probe opened' }
  })

  on('ui.render', { component: 'Pane', requestId: PANE }, async ($, e) => {
    const ui = $.ui.resolve(e) as ElementTable<'desktop'>
    const { Box, Button, Svg, Text } = ui
    const rows = Array.from({ length: ROWS }, (_, i) => i)

    return (
      <Box flexDirection="column">
        <Box flexDirection="row" gap={2}>
          <Button
            key="to-250"
            label="Scroll to row 250"
            onPress={() => {
              void $.ui.scroll({ in: PANE, to: { key: 'row-250' } })
            }}
          />
          <Button
            key="to-0"
            label="Scroll to row 0"
            onPress={() => {
              void $.ui.scroll({ in: PANE, to: { key: 'row-0' } })
            }}
          />
        </Box>
        {rows.map(i => (
          <Box key={`row-${i}`} flexDirection="row" alignItems="center">
            {HEIGHTS.map(height => (
              <Svg source={lineSvg(height)} alt={`probe ${height}px`} />
            ))}
            <Box flexGrow={1} overflow="hidden">
              <Button key={`commit-${i}`} plain onPress={() => $.ui.toast(`pressed ${i}`)}>
                <Text>{`row ${i} - click target`}</Text>
              </Button>
            </Box>
          </Box>
        ))}
      </Box>
    )
  })
}
```

```ts
const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']

const pad = (value: number): string => String(value).padStart(2, '0')

export function shortHash(hash: string): string {
  return hash.slice(0, 8)
}

export function formatDate(timestamp: number): string {
  const date = new Date(timestamp * 1000)
  const month = MONTHS[date.getMonth()] ?? ''

  return `${pad(date.getDate())} ${month} ${date.getFullYear()} ${pad(date.getHours())}:${pad(date.getMinutes())}`
}
```

```ts
import type { ViewState } from '../types'

type BranchView = Pick<ViewState, 'branches' | 'remoteBranches' | 'isRemoteShown'>

export function visibleBranches(view: BranchView): string[] {
  if (view.isRemoteShown) {
    return [...view.branches]
  }

  const remotes = new Set(view.remoteBranches)

  return view.branches.filter(name => !remotes.has(name))
}
```

`hooks/git/parse-log.test.ts`:

```ts
import { describe, expect, test } from 'claude-code/testing'

import { parseLog } from './parse-log'

const A = 'a'.repeat(40)
const B = 'b'.repeat(40)
const C = 'c'.repeat(40)

const record = (...fields: string[]): string => `${fields.join('\x1f')}\x1e\n`

describe('parseLog', () => {
  test('parses hash, parents, author, time, refs and subject', () => {
    const stdout = record(A, `${B} ${C}`, 'Ada', '1760000000', 'HEAD -> main, origin/main', 'feat: add | pipe')

    expect(parseLog(stdout, new Set(['origin/main']))).toEqual([
      {
        hash: A,
        parents: [B, C],
        author: 'Ada',
        timestamp: 1760000000,
        subject: 'feat: add | pipe',
        refs: [
          { name: 'main', kind: 'branch', isCurrent: true },
          { name: 'origin/main', kind: 'remote', isCurrent: false },
        ],
      },
    ])
  })

  test('a root commit has no parents', () => {
    const [commit] = parseLog(record(A, '', 'Ada', '1', '', 'init'), new Set())

    expect(commit?.parents).toEqual([])
  })

  test('keeps non-ASCII subjects intact', () => {
    const [commit] = parseLog(record(A, B, '悟空', '1', '', 'docs: 添加 PRD'), new Set())

    expect(commit?.author).toBe('悟空')
    expect(commit?.subject).toBe('docs: 添加 PRD')
  })

  test('parses several records in order', () => {
    const stdout = record(A, B, 'x', '2', '', 's1') + record(B, '', 'x', '1', '', 's2')

    expect(parseLog(stdout, new Set()).map(commit => commit.hash)).toEqual([A, B])
  })

  test('empty output gives no commits', () => {
    expect(parseLog('', new Set())).toEqual([])
    expect(parseLog('\n', new Set())).toEqual([])
  })
})
```

```ts
import type { Ref } from '../../types'

const HEAD_ARROW = 'HEAD -> '
const TAG_PREFIX = 'tag: '

export function parseRefs(raw: string, remoteNames: ReadonlySet<string>): Ref[] {
  const text = raw.trim()

  if (text === '') {
    return []
  }

  return text.split(', ').map((part): Ref => {
    const item = part.trim()

    if (item.startsWith(HEAD_ARROW)) {
      return { name: item.slice(HEAD_ARROW.length), kind: 'branch', isCurrent: true }
    }

    if (item === 'HEAD') {
      return { name: 'HEAD', kind: 'head', isCurrent: true }
    }

    if (item.startsWith(TAG_PREFIX)) {
      return { name: item.slice(TAG_PREFIX.length), kind: 'tag', isCurrent: false }
    }

    const isRemote = remoteNames.has(item) || item.endsWith('/HEAD')

    return { name: item, kind: isRemote ? 'remote' : 'branch', isCurrent: false }
  })
}
```

```ts
const FIELD = '\x1f'
const RECORD = '\x1e'

export const LOG_FORMAT = ['%H', '%P', '%an', '%at', '%D', '%s'].join('%x1f') + '%x1e'

export function parseLog(stdout: string, remoteNames: ReadonlySet<string>): Commit[] {
  const commits: Commit[] = []

  for (const record of stdout.split(RECORD)) {
    const body = record.replace(/^\n+/, '')

    if (body.trim() === '') {
      continue
    }

    const [hash, parents, author, timestamp, refs, subject] = body.split(FIELD)

    if (
      hash === undefined ||
      parents === undefined ||
      author === undefined ||
      timestamp === undefined ||
      refs === undefined ||
      subject === undefined
    ) {
      continue
    }

    commits.push({
      hash,
      parents: parents.split(' ').filter(parent => parent !== ''),
      author,
      timestamp: Number(timestamp),
      subject,
      refs: parseRefs(refs, remoteNames),
    })
  }

  return commits
}
```

```ts
import type { EngineInterface } from 'claude-code'

const READ_ONLY = new Set(['log', 'show', 'for-each-ref', 'rev-parse'])
const HASH_PATTERN = /^[0-9a-f]{40,64}$/
const FIXED_OPTIONS = ['-c', 'core.quotepath=false']

export type GitResult =
  | { ok: true; stdout: string }
  | {
      ok: false
      kind: 'not-a-repo' | 'no-commits' | 'truncated' | 'unavailable' | 'failed'
      message: string
    }

export type GitFailure = Extract<GitResult, { ok: false }>

export function buildGitArgv(args: readonly string[]): string[] {
  const [sub] = args

  if (sub === undefined || !READ_ONLY.has(sub)) {
    throw new Error(`git ${sub ?? ''} is not an allowed read-only command`)
  }

  return ['git', ...FIXED_OPTIONS, ...args]
}

export function isCommitHash(value: string): boolean {
  return HASH_PATTERN.test(value)
}

export async function runGit($: EngineInterface, args: readonly string[]): Promise<GitResult> {
  const argv = buildGitArgv(args)

  let result: Awaited<ReturnType<EngineInterface['process']['run']>>

  try {
    result = await $.process.run(argv)
  } catch (error) {
    return { ok: false, kind: 'unavailable', message: error instanceof Error ? error.message : String(error) }
  }

  if (result.isStdoutTruncated) {
    return { ok: false, kind: 'truncated', message: 'Git output was too large to read in one page.' }
  }

  if (result.exitCode === 0) {
    return { ok: true, stdout: result.stdout }
  }

  if (/not a git repository/i.test(result.stderr)) {
    return { ok: false, kind: 'not-a-repo', message: 'Not a git repository.' }
  }

  if (/does not have any commits yet|unknown revision|ambiguous argument 'HEAD'/i.test(result.stderr)) {
    return { ok: false, kind: 'no-commits', message: 'The repository has no commits yet.' }
  }

  const firstLine = result.stderr.split('\n')[0]?.trim() ?? ''

  return { ok: false, kind: 'failed', message: firstLine === '' ? `git exited with ${result.exitCode}` : firstLine }
}
```

```ts
      stdout: reply.stdout ?? '',
      stderr: reply.stderr ?? '',
      isStdoutTruncated: reply.isStdoutTruncated ?? false,

    }
  })
```

```ts
import type { Edge, GraphRow, LaneState } from '../../types'

export type LayoutInput = { hash: string; parents: readonly string[] }

export function emptyLaneState(): LaneState {
  return { lanes: [], colors: [], nextColor: 0 }
}

function firstFree(lanes: readonly (string | null)[]): number {
  const free = lanes.indexOf(null)

  return free === -1 ? lanes.length : free
}

function colorAt(colors: readonly number[], lane: number): number {
  return colors[lane] ?? 0
}

function trim(lanes: (string | null)[], colors: number[]): void {
  while (lanes.length > 0 && lanes[lanes.length - 1] === null) {
    lanes.pop()
    colors.pop()
  }
}

export function layoutCommits(
  prev: LaneState,
  commits: readonly LayoutInput[],
): { rows: GraphRow[]; state: LaneState } {
  const lanes = [...prev.lanes]
  const colors = [...prev.colors]
  let nextColor = prev.nextColor
  const rows: GraphRow[] = []

  for (const commit of commits) {
    const before = [...lanes]
    const colorsBefore = [...colors]
    const waiting: number[] = []

    before.forEach((hash, index) => {
      if (hash === commit.hash) {
        waiting.push(index)
      }
    })

    let lane: number
    let color: number

    if (waiting.length > 0) {
      lane = Math.min(...waiting)
      color = colorAt(colorsBefore, lane)
    } else {
      lane = firstFree(lanes)
      color = nextColor
      nextColor += 1
    }

    const incoming: Edge[] = waiting.map(from => ({ from, to: lane, color: colorAt(colorsBefore, from) }))
    const through: Edge[] = []

    before.forEach((hash, index) => {
      if (hash !== null && !waiting.includes(index)) {
        through.push({ from: index, to: index, color: colorAt(colorsBefore, index) })
      }
    })

    for (const index of waiting) {
      lanes[index] = null
    }

    const outgoing: Edge[] = []

    commit.parents.forEach((parent, order) => {
      const existing = lanes.indexOf(parent)

      if (existing !== -1) {
        outgoing.push({ from: lane, to: existing, color: colorAt(colors, existing) })
      } else if (order === 0) {
        lanes[lane] = parent
        colors[lane] = color
        outgoing.push({ from: lane, to: lane, color })
      } else {
        const slot = firstFree(lanes)
        const slotColor = nextColor
        nextColor += 1
        lanes[slot] = parent
        colors[slot] = slotColor
        outgoing.push({ from: lane, to: slot, color: slotColor })
      }
    })

    trim(lanes, colors)

    rows.push({
      hash: commit.hash,
      lane,
      color,
      isMerge: commit.parents.length > 1,
      through,
      incoming,
      outgoing,
    })
  }

  return { rows, state: { lanes, colors, nextColor } }
}

export function columnCount(rows: readonly GraphRow[]): number {
  let widest = 1

  for (const row of rows) {
    widest = Math.max(widest, row.lane + 1)

    for (const edge of [...row.through, ...row.incoming, ...row.outgoing]) {
      widest = Math.max(widest, edge.from + 1, edge.to + 1)
    }
  }

  return widest
}
```

```ts
import type { Edge, GraphRow } from '../../types'
import { LANE_COLORS, LANE_WIDTH, LINE_WIDTH, NODE_RADIUS, ROW_HEIGHT } from '../constants'

const MID = ROW_HEIGHT / 2
const FALLBACK_COLOR = '#8b949e'

const laneX = (lane: number): number => lane * LANE_WIDTH + LANE_WIDTH / 2

const colorOf = (index: number): string => LANE_COLORS[index % LANE_COLORS.length] ?? FALLBACK_COLOR

const stroke = (color: number): string =>
  `fill="none" stroke="${colorOf(color)}" stroke-width="${LINE_WIDTH}" stroke-linecap="round"`

function throughLine(edge: Edge): string {
  const x = laneX(edge.from)

  return `<line x1="${x}" y1="0" x2="${x}" y2="${ROW_HEIGHT}" ${stroke(edge.color)}/>`
}

function incomingLine(edge: Edge, nodeLane: number): string {
  const from = laneX(edge.from)
  const to = laneX(nodeLane)

  if (from === to) {
    return `<line x1="${from}" y1="0" x2="${to}" y2="${MID}" ${stroke(edge.color)}/>`
  }

  return `<path d="M${from} 0 C${from} ${MID} ${to} 0 ${to} ${MID}" ${stroke(edge.color)}/>`
}

function outgoingLine(edge: Edge, nodeLane: number): string {
  const from = laneX(nodeLane)
  const to = laneX(edge.to)

  if (from === to) {
    return `<line x1="${from}" y1="${MID}" x2="${to}" y2="${ROW_HEIGHT}" ${stroke(edge.color)}/>`
  }

  return `<path d="M${from} ${MID} C${from} ${ROW_HEIGHT} ${to} ${MID} ${to} ${ROW_HEIGHT}" ${stroke(edge.color)}/>`
}

function node(row: GraphRow, isHead: boolean): string {
  const x = laneX(row.lane)
  const color = colorOf(row.color)

  if (isHead) {
    return `<circle cx="${x}" cy="${MID}" r="${NODE_RADIUS}" fill="none" stroke="${color}" stroke-width="${LINE_WIDTH}"/>`
  }

  return `<circle cx="${x}" cy="${MID}" r="${NODE_RADIUS}" fill="${color}"/>`
}

export function rowSvg(row: GraphRow, columns: number, isHead: boolean): string {
  const width = Math.max(columns, 1) * LANE_WIDTH

  const parts = [
    ...row.through.map(throughLine),
    ...row.incoming.map(edge => incomingLine(edge, row.lane)),
    ...row.outgoing.map(edge => outgoingLine(edge, row.lane)),
    node(row, isHead),
  ]

  return (
    `<svg xmlns="http://www.w3.org/2000/svg" width="${width}" height="${ROW_HEIGHT}" ` +
    `viewBox="0 0 ${width} ${ROW_HEIGHT}">${parts.join('')}</svg>`
  )
}
```

```ts
import { read, update } from 'claude-code'
import type { EngineInterface } from 'claude-code'

import type { Commit, ViewState } from '../types'
import { visibleBranches } from './branches'
import { ALL_BRANCHES, PAGE_SIZE } from './constants'
import { LOG_FORMAT, parseLog } from './git/parse-log'
import { parseBranches } from './git/parse-branches'
import { META_FORMAT, parseMeta, parseNumstat } from './git/parse-detail'
import { isCommitHash, runGit } from './git/run-git'
import type { GitFailure } from './git/run-git'
import { emptyLaneState, layoutCommits } from './layout/lanes'
import { viewAtom } from './state'

type Page = { commits: Commit[]; hasMore: boolean }
type PageResult = { ok: true; page: Page } | GitFailure

function buildRevs(view: ViewState): string[] | null {
  if (view.branch === ALL_BRANCHES) {
    return ['--branches', '--tags', ...(view.isRemoteShown ? ['--remotes'] : []), 'HEAD']
  }

  return visibleBranches(view).includes(view.branch) ? [view.branch] : null
}

async function fetchPage($: EngineInterface, view: ViewState, offset: number): Promise<PageResult> {
  const revs = buildRevs(view)

  if (revs === null) {
    return { ok: false, kind: 'failed', message: 'That branch no longer exists.' }
  }

  const result = await runGit($, [
    'log',
    '--topo-order',
    `--skip=${offset}`,
    '-n',
    String(PAGE_SIZE + 1),
    `--format=${LOG_FORMAT}`,
    ...revs,
    '--',
  ])

  if (!result.ok) {
    return result.kind === 'no-commits' ? { ok: true, page: { commits: [], hasMore: false } } : result
  }

  const parsed = parseLog(result.stdout, new Set(view.remoteBranches))

  return { ok: true, page: { commits: parsed.slice(0, PAGE_SIZE), hasMore: parsed.length > PAGE_SIZE } }
}

async function fail($: EngineInterface, failure: GitFailure): Promise<void> {
  await update($, viewAtom, view => ({
    ...view,
    status: failure.kind === 'not-a-repo' ? 'no-repo' : 'error',
    errorMessage: failure.message,
  }))
}

export async function loadFirst($: EngineInterface): Promise<void> {
  const current = await read($, viewAtom)
  await update($, viewAtom, view => ({ ...view, status: 'loading', errorMessage: '' }))

  const inside = await runGit($, ['rev-parse', '--is-inside-work-tree'])

  if (!inside.ok) {
    return fail($, inside)
  }

  const listed = await runGit($, ['for-each-ref', '--format=%(refname)', 'refs/heads', 'refs/remotes'])

  if (!listed.ok) {
    return fail($, listed)
  }

  const { locals, remotes } = parseBranches(listed.stdout)
  const remoteBranches = remotes.filter(name => !name.endsWith('/HEAD'))
  const branches = [...locals, ...remoteBranches]
  const probe: ViewState = { ...current, branches, remoteBranches }
  const branch =
    current.branch === ALL_BRANCHES || visibleBranches(probe).includes(current.branch) ? current.branch : ALL_BRANCHES

  const result = await fetchPage($, { ...probe, branch }, 0)

  if (!result.ok) {
    return fail($, result)
  }

  const { rows, state } = layoutCommits(emptyLaneState(), result.page.commits)

  await update($, viewAtom, view => ({
    ...view,
    status: 'ready',
    errorMessage: '',
    branches,
    remoteBranches,
    branch,
    commits: result.page.commits,
    rows,
    laneState: state,
    hasMore: result.page.hasMore,
    expandedHash: null,
    detail: null,
    detailError: '',
  }))
}

export async function loadMore($: EngineInterface): Promise<void> {
  const view = await read($, viewAtom)

  if (!view.hasMore || view.status !== 'ready') {
    return
  }

  const result = await fetchPage($, view, view.commits.length)

  if (!result.ok) {
    await update($, viewAtom, latest => ({ ...latest, errorMessage: result.message }))

    return
  }

  const known = new Set(view.commits.map(commit => commit.hash))
  const fresh = result.page.commits.filter(commit => !known.has(commit.hash))
  const { rows, state } = layoutCommits(view.laneState, fresh)

  await update($, viewAtom, latest => {
    if (latest.branch !== view.branch || latest.commits.length !== view.commits.length) {
      return latest
    }

    return {
      ...latest,
      errorMessage: '',
      commits: [...latest.commits, ...fresh],
      rows: [...latest.rows, ...rows],
      laneState: state,
      hasMore: result.page.hasMore,
    }
  })
}

const clearedList = (view: ViewState): ViewState => ({
  ...view,
  commits: [],
  rows: [],
  laneState: emptyLaneState(),
  hasMore: false,
  expandedHash: null,
  detail: null,
  detailError: '',
})

export async function selectBranch($: EngineInterface, value: string): Promise<void> {
  const view = await read($, viewAtom)

  if (value !== ALL_BRANCHES && !visibleBranches(view).includes(value)) {
    return
  }

  await update($, viewAtom, latest => ({ ...clearedList(latest), branch: value }))
  await loadFirst($)
}

export async function toggleRemote($: EngineInterface): Promise<void> {
  const view = await read($, viewAtom)
  const isRemoteShown = !view.isRemoteShown
  const isSelectionHidden = !isRemoteShown && view.remoteBranches.includes(view.branch)

  await update($, viewAtom, latest => ({
    ...clearedList(latest),
    isRemoteShown,
    branch: isSelectionHidden ? ALL_BRANCHES : latest.branch,
  }))
  await loadFirst($)
}

export async function refresh($: EngineInterface): Promise<void> {
  await loadFirst($)
}

export async function toggleCommit($: EngineInterface, hash: string): Promise<void> {
  const view = await read($, viewAtom)

  if (view.expandedHash === hash) {
    await update($, viewAtom, latest => ({ ...latest, expandedHash: null, detail: null, detailError: '' }))

    return
  }

  if (!isCommitHash(hash)) {
    return
  }

  await update($, viewAtom, latest => ({ ...latest, expandedHash: hash, detail: null, detailError: '' }))

  const meta = await runGit($, ['show', '-s', `--format=${META_FORMAT}`, hash])
  const files = await runGit($, ['show', '-m', '--first-parent', '--numstat', '--format=', hash])
  const parsed = meta.ok ? parseMeta(meta.stdout) : null
  const error = meta.ok ? (parsed === null ? 'Could not read the commit details.' : '') : meta.message
  const detail = parsed === null ? null : { ...parsed, files: files.ok ? parseNumstat(files.stdout) : [] }

  await update($, viewAtom, latest =>
    latest.expandedHash === hash ? { ...latest, detail, detailError: error } : latest,
  )
}
```

```ts
import type { EngineInterface } from 'claude-code'

import { PANE_ID } from '../constants'
import { loadMore, refresh, selectBranch, toggleCommit, toggleRemote } from '../controller'

export type Actions = {
  selectBranch(value: string): void
  toggleRemote(): void
  refresh(): void
  loadMore(): void
  toggleCommit(hash: string): void
  scrollToCommit(hash: string): void
}

export function makeActions($: EngineInterface): Actions {
  const guard = (task: Promise<unknown>): void => {
    task.catch(error => {
      $.ui.toast(`git-graph: ${error instanceof Error ? error.message : String(error)}`)
    })
  }

  return {
    selectBranch: value => guard(selectBranch($, value)),
    toggleRemote: () => guard(toggleRemote($)),
    refresh: () => guard(refresh($)),
    loadMore: () => guard(loadMore($)),
    toggleCommit: hash => guard(toggleCommit($, hash)),
    scrollToCommit: hash => guard($.ui.scroll({ in: PANE_ID, to: { key: `row-${hash}` } })),
  }
}
```

```tsx
import type { ElementTable } from 'claude-code'

import type { ViewState } from '../../types'
import { AUTHOR_WIDTH, CELL_WIDTH_PX, DATE_WIDTH, HASH_WIDTH, LANE_WIDTH } from '../constants'
import { columnCount } from '../layout/lanes'
import type { Actions } from './actions'
import { renderDetail } from './commit-detail'
import { renderRow } from './commit-row'
import { renderToolbar } from './toolbar'

type DesktopUi = ElementTable<'desktop'>

function renderHeader(ui: DesktopUi, columns: number) {
  const { Box, Text } = ui
  const graphCells = Math.ceil((columns * LANE_WIDTH) / CELL_WIDTH_PX)

  return (
    <Box flexDirection="row">
      <Box width={graphCells}>
        <Text bold>Graph</Text>
      </Box>
      <Box flexGrow={1}>
        <Text bold>Description</Text>
      </Box>
      <Box width={DATE_WIDTH}>
        <Text bold>Date</Text>
      </Box>
      <Box width={AUTHOR_WIDTH}>
        <Text bold>Author</Text>
      </Box>
      <Box width={HASH_WIDTH}>
        <Text bold>Commit</Text>
      </Box>
    </Box>
  )
}

function renderList(ui: DesktopUi, view: ViewState, actions: Actions) {
  const { Box, Button, Text } = ui
  const columns = columnCount(view.rows)
  const loaded = new Set(view.commits.map(commit => commit.hash))

  const items = view.commits.flatMap((commit, index) => {
    const row = view.rows[index]

    if (row === undefined) {
      return []
    }

    const parts = [renderRow(ui, commit, row, columns, actions)]

    if (view.expandedHash === commit.hash) {
      parts.push(renderDetail(ui, view, loaded, actions))
    }

    return parts
  })

  return (
    <Box flexDirection="column">
      {renderHeader(ui, columns)}
      {items}
      {view.errorMessage === '' ? null : <Text color="error">{view.errorMessage}</Text>}
      {view.hasMore ? <Button key="more" label="加载更多" onPress={() => actions.loadMore()} /> : null}
    </Box>
  )
}

function renderBody(ui: DesktopUi, view: ViewState, actions: Actions) {
  const { Text } = ui
  const isLoading = view.status === 'idle' || (view.status === 'loading' && view.commits.length === 0)

  if (isLoading) {
    return <Text dimColor>Loading…</Text>
  }

  if (view.status === 'no-repo') {
    return <Text>当前目录不是 Git 仓库。</Text>
  }

  if (view.status === 'error') {
    return <Text color="error">{`无法读取 Git 历史:${view.errorMessage}`}</Text>
  }

  if (view.commits.length === 0) {
    return <Text dimColor>这个仓库还没有提交。</Text>
  }

  return renderList(ui, view, actions)
}

export function renderPane(ui: DesktopUi, view: ViewState, actions: Actions) {
  const { Box } = ui

  return (
    <Box flexDirection="column" gap={1}>
      {renderToolbar(ui, view, actions)}
      {renderBody(ui, view, actions)}
    </Box>
  )
}
```
